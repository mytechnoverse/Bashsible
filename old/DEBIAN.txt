    bashutils_error_list+=("try function requires an error message followed by a function or command or script name")
    return "$false"
  }
  bashutils_error_list+=("$1")
  shift
  "$@"
  declare "$integer_variable_option" bashutils_exit_code="$?"
  (($bashutils_exit_code != 0)) && return "$bashutils_exit_code"
  unset 'bashutils_error_list[-1]'
  bashutils_error_list=("${bashutils_error_list[@]}")
}

parameter_count() {
  declare "$integer_variable_option" target_parameter_count="$1"
  declare "$integer_variable_option" parameter_count="$#"
  parameter_count=$((parameter_count - 1))
  true_condition=$(( target_parameter_count == parameter_count ))
  declare parent_function_name="${FUNCNAME[$parent_function_index]}"
  try "${parent_function_name} function requires ${target_parameter_count} parameter(s)" (( ! true_condition ))
}

parameter_count_minimum() {
  declare "$integer_variable_option" minimum_parameter_count="$1"
  declare "$integer_variable_option" parameter_count="$#"
  parameter_count=$((parameter_count - 1))
  true_condition=$(( minimum_parameter_count <= parameter_count ))
  declare parent_function_name="${FUNCNAME[$parent_function_index]}"
  try "${parent_function_name} function requires at least ${minimum_parameter_count} parameter(s)" (( ! true_condition ))
}

ADD_STATIC_DNS_RECORD() {
  local DOMAIN_NAME="$1"
  local A_RECORD="$2"
  local HOSTS_FILE_PATH="/etc/hosts"
  local ACTIVE_LINES=$(grep -v -E '^[[:space:]]*#.*$' "$HOSTS_FILE_PATH")
  local TARGET_LINES=$(printf '%s\n' "$ACTIVE_LINES" | grep -E '[[:space:]]'"$DOMAIN_NAME"'([[:space:]]|$)')
  if [[ -n "$TARGET_LINES" ]]
  then
    local TARGET_LINES_COUNT=$(printf '%s\n' "$TARGET_LINES" | wc -l)
    [[ $TARGET_LINES_COUNT -eq 1 ]] || {
      ERROR_MSG="${DOMAIN_NAME} DUPLICATE Entries"
      return 1
    }
    local DOMAIN_REGEX=${DOMAIN_NAME//./\\.}
    if ! printf '%s\n' "$TARGET_LINES" | grep -q -E "^[[:space:]]*${A_RECORD//./\\.}[[:space:]].*[[:space:]]'"$DOMAIN_REGEX"'([[:space:]]|$)"
    then
      ERROR_MSG="${DOMAIN_NAME} Entry SYNTAX"
      return 1
    fi
  else
    printf '%s\n' "${A_RECORD} ${DOMAIN_NAME}" | sudo tee -a "$HOSTS_FILE_PATH" > /dev/null
  fi
}

CONFIG_HOSTS() {
  local CFG_MODE="$1"
  local NEW_HOSTNAME="$2"
  local SITE_DOMAIN="$3"
  local CONF_HOSTS_ENTRY=""
  if [[ -z "$SITE_DOMAIN" ]]
  then
    CONF_HOSTS_ENTRY="$NEW_HOSTNAME"
  else
    CONF_HOSTS_ENTRY="${NEW_HOSTNAME}.${SITE_DOMAIN} ${NEW_HOSTNAME}"
  fi
  local CONF_HOSTS_PATH="/etc/hosts"
  CONF_READABLE "$CONF_HOSTS_PATH" || return 1
  SUDO CONF_WRITABLE "$CONF_HOSTS_PATH" || return 1
  if [[ "$CFG_MODE" eq "replace" ]]
  then
    CONF_HOSTS_CONTENT=""
    CONF_HOSTS_GENERATE
    printf "%b" "$CONF_HOSTS_CONTENT" | SUDO tee "$CONF_HOSTS_PATH" >/dev/null
  elif [[ "$CFG_MODE" eq "update" ]]
  then
    CONF_HOSTS_UPDATE
  fi
}

DIR_ACCESSIBLE() {
  local DIR_PATH="$1"
  [[ -n "$DIR_PATH" && -x "$DIR_PATH" ]] || {
    ERROR_MSG="${DIR_PATH} Directory EXECUTE Permission"
    return 1
  }
}

PARENT_DIR_ACCESSIBLE() {
  local DIR_PATH="$1"
  local PARENT_DIR_PATH="$(dirname "$DIR_PATH")"
  DIR_ACCESSIBLE "$PARENT_DIR_PATH" || return 1
}



DIR_PRESENT() {
  local DIR_PATH="$1"
  PARENT_DIR_ACCESSIBLE "$DIR_PATH" || return 1
  [[ -d "$DIR_PATH" ]] || {
    ERROR_MSG="${DIR_PATH:-<empty>} Directory NOT Found"
    return 1
  }
}

DIR_READABLE() {
  local DIR_PATH="$1"
  PARENT_DIR_ACCESSIBLE "$DIR_PATH" || return 1
  [[ -r "$DIR_PATH" ]] || {
    ERROR_MSG="${DIR_PATH} Directory READ Permission"
    return 1
  }
}

APP_INSTALLED() {
  local APP_NAME="$1"
  [[ -n "$APP_NAME" ]] || {
    ERROR_MSG="Application Not Specified"
    return 1
  }
  command -v "$APP_NAME" >/dev/null 2>&1 || {
    ERROR_MSG="${APP_NAME} Application Not Installed"
    return 1
  }
}

PROMPT_SUDO() {
  if [[ $EUID -ne 0 ]]
  then
    APP_INSTALLED sudo || return 1
    sudo -v || {
      ERROR_MSG="SUDO Authentication Failed"
      return 1
    }
  fi
}

SUDO() {
  if [[ $EUID -eq 0 ]]
  then
    "$@"
  else
    sudo "$@"
  fi
}

APP_AVAILABLE() {
  local APP_NAME="$1"
  [[ -n "$APP_NAME" ]] || return 1
  if ! APP_INSTALLED "$APP_NAME" && ! FILE_EXECUTABLE "./$APP_NAME"
  then
    ERROR_MSG="${APP_NAME} Application Not Available"
    return 1
  fi
}

    fi

UPDATE_PKGS_INDEX() {
  [[ "${PKGS_INDEX_UPDATED:-}" == "true" ]] || {
    PROMPT_SUDO || return 1
    export DEBIAN_FRONTEND="noninteractive"
    export NEEDRESTART_MODE="a"
    if SUDO apt-get update >/dev/null
    then
      export PKGS_INDEX_UPDATED=true
    else
      ERROR_MSG="APT Packages Index Update Failed"
      return 1
    fi
  }
}

INSTALL_PKG() {
  local PKG_NAME="$1"
  [[ -n "$PKG_NAME" ]] || {
    ERROR_MSG="Package Not Specified"
    return 1
  }
  PKG_INSTALLED "$PKG_NAME" || {
    PROMPT_SUDO || return 1
    export DEBIAN_FRONTEND="noninteractive"
    export NEEDRESTART_MODE="a"
    SUDO apt-get install -y "$PKG_NAME" >/dev/null || {
      ERROR_MSG="${PKG_NAME} Package Installation Failed"
      return 1
    }
  }
}

  }

PROVIDE_APPS() {
  local -n APPS_LIST=$1
  local -n PKGS_LIST=$2
  local INDEX_UPDATED=false
  for APP_INDEX in "${!APPS_LIST[@]}"
  do
    local APP_NAME="${APPS_LIST[$APP_INDEX]}"
    if ! CHECK_APP "$APP_NAME"
    then
      local PKG_NAME="${PKGS_LIST[$APP_INDEX]}"
      if ! CHECK_PKG "$PKG_NAME"
      then
        if [[ "$INDEX_UPDATED" == "false" ]]
        then
          if ! UPDATE_PKG_INDEX
          then
            ERROR_MSG="Package Index Update Failed"
            return 1
          else
            INDEX_UPDATED=true
          fi
        fi
        if ! INSTALL_PKG "$PKG_NAME"
        then
          ERROR_MSG="$PKG_NAME Package Installation Failed"
          return 1
        fi
      fi
    fi
  done
}
