(
    set -Eeuo pipefail
    shopt -s inherit_errexit
    [[ -v apt_index_update_status ]] || apt_index_update_status=0

    declare -A apt_repo_dependencies_official_dictionary=(
        [gpgv]='gpgv|'
        [debian-archive-keyring]='|'
    )

    declare -A apt_repo_dependencies_external_dictionary=(
        [curl]='curl|'
        [ca-certificates]='|'
        [gpg]='gpg|--no-install-recommends'
    )

    ensure_application() {
        local -n apt_index_update_status_ref=$1
        local this_application_package=$2
        local this_application_binary=${3-}
        local this_application_package_options=${4-}
        [[ $(dpkg-query --show --showformat='${Status}' $this_application_package 2>/dev/null) == 'install ok installed' ]] || {
            (( apt_index_update_status_ref )) || {
                apt-get --quiet=2 -o Acquire::Retries=0 -o Acquire::http::Timeout=300 -o Acquire::https::Timeout=300 --error-on=any update || {
                    printf '%s\n' "apt index update failed error"
                    return 1
                }
                apt_index_update_status_ref=1
            }
            local -a this_application_package_options_list=()
            [[ -z "$this_application_package_options" ]] || {
                read -r -a this_application_package_options_list <<< "$this_application_package_options"
            }
            DEBIAN_FRONTEND=noninteractive apt-get --quiet=2 \
            -o Acquire::Retries=0 -o Acquire::http::Timeout=300 -o Acquire::https::Timeout=300 \
            --assume-yes "${this_application_package_options_list[@]}" install $this_application_package || {
                printf '%s\n' "package ( ${this_application_package} ) install error"
                return 1
            }
        }
        [[ -z "$this_application_binary" ]] || {
            command -v $this_application_binary >/dev/null || {
                printf '%s\n' "binary ( ${this_application_binary} ) not found in PATH error"
                return 1
            }
        }
    }

    ensure_dependencies_dictionary() {
        local -n apt_index_update_status_ref=$1
        local -n dependencies_dictionary=$2
        for this_application_package in "${!dependencies_dictionary[@]}"
        do
            local this_application_properties="${dependencies_dictionary[$this_application_package]}"
            local this_application_binary
            local this_application_package_options
            IFS='|' read -r this_application_binary this_application_package_options <<< "$this_application_properties"
            ensure_application "${!apt_index_update_status_ref}" $this_application_package $this_application_binary "$this_application_package_options"
        done
    }
)




exit

        ssl_conf_command Ciphersuites TLS_AES_128_GCM_SHA256;
        proxy_http_version 1.1;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
    }
    server {
        listen 80;
        server_name ${nginx_service_name};
        return 301 https://\$host\$request_uri;
    }
$(printf '\x1c')
